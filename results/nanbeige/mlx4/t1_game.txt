let audioCtx = null, muted = false;
function beep(freq, dur=0.08, type='square', vol=0.06, slide=0){
  if (muted) return;
  try {
    audioCtx = audioCtx || new (window.AudioContext||window.webkitAudioContext)();
    const o = audioCtx.createOscillator(), g = audioCtx.createGain();
    o.type = type; o.frequency.value = freq;
    if (slide) o.frequency.exponentialRampToValueAtTime(freq*slide, audioCtx.currentTime+dur);
    g.gain.value = vol;
    o.connect(g).connect(audioCtx.destination);

}

function updateEffects(dt){
  for (let i=particles.length-1;i>=0;i--){
    const p = particles[i];
    p.x += p.vx*dt; p.y += p.vy*dt; p.life -= dt/p.maxLife;
    if (p.life<=0) particles.splice(i,1);
  }
  for (let i=popups.length-1;i>=0;i--){

}

fitCanvas();

let state = 'ready';               // ready | playing | over
let particles = [], popups = [];

try { best = parseInt(localStorage.getItem('snake.best')) || 0; } catch(e){ best = 0; }

/* ═══ Sound (tiny synth, no assets) ═══ */
let audioCtx = null, muted = false;
function beep(freq, dur, type, vol, slide){
  if (muted) return;
  try{
    audioCtx = audioCtx || new (window.AudioContext || window.webkitAudioContext)();
    if (audioCtx.state === 'suspended') audioCtx.resume();
    const t0 = audioCtx.currentTime;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type || 'square';
    osc.frequency.setValueAtTime(freq, t0);
    if (slide) osc.frequency.exponentialRampToValueAtTime(Math.max(40, freq*slide), t0+dur);
    gain.gain.setValueAtTime(vol || .05, t0);
    gain.gain.exponentialRampToValueAtTime(.0001, t0+dur);
    osc.connect(gain); gain.connect(audioCtx.destination);
    osc.start(t0); osc.stop(t0+dur+.03);
  }catch(e){}
}
soundBtn.addEventListener('click', ()=>{
  muted = !muted;
  soundBtn.textContent = muted ? '🔇' : '🔊';
  soundBtn.classList.toggle('off', muted);
});

/* ═══ Core helpers ═══ */
function currentSpeed(){ return Math.max(MIN_SPEED, BASE_SPEED - (speedLevel-1)*SPEED_STEP); }

function spawnFood(){
  const free = [];
  for (let y=0; y<GRID; y++)
    for (let x=0; x<GRID; x++)
      if (!snake.some(s => s.x===x && s.y===y)) free.push({x,y});
  food = free.length ? free[(Math.random()*free.length)|0] : {x:-1,y:-1};
}

function reset(){
  const cx = Math.floor(GRID/2), cy = Math.floor(GRID/2);
  snake = [{x:cx,y:cy},{x:cx-1,y:cy},{x:cx-2,y:cy},{x:cx-3,y:cy}];
  dir = {x:1,y:0};
  queue = [];
  score = 0;
  speedLevel = 1;
  newBest = false;
  particles = [];
  popups = [];
  shake = 0;
  flash = 0;
  spawnFood();
  updateHUD();
}

function updateHUD(){
  elScore.textContent = score;
  elBest.textContent  = best;
  elSpeed.textContent = 'LV ' + speedLevel;
}

function queueDir(d){
  const last = queue.length ? queue[queue.length-1] : dir;
  // no 180° reversal, no duplicate entries
  if ((d.x === -last.x && d.y === -last.y) || (d.x === last.x && d.y === last.y)) return;
  if (queue.length < 2) queue.push(d);
}

/* ═══ Game step ═══ */
function step(){
  const d = queue.length ? queue.shift() : dir;
  dir = d;
  const head = { x: snake[0].x + dir.x, y: snake[0].y + dir.y };

  // wall collision
  if (head.x < 0 || head.y < 0 || head.x >= GRID || head.y >= GRID){
    return gameOver('You hit the wall', 'over');
  }

  const eating = (head.x === food.x && head.y === food.y);
  // self collision — the tail cell is safe unless we're growing
  const checkLen = eating ? snake.length : snake.length - 1;
  for (let i=0; i<checkLen; i++){
    if (snake[i].x === head.x && snake[i].y === head.y){
      return gameOver('You hit yourself', 'over');
    }
  }

  snake.unshift(head);

  if (eating){
    score += 1;
    const fx = food.x*CELL + CELL/2, fy = food.y*CELL + CELL/2;
    spawnParticles(fx, fy, '#ff6b5e', 16);
    popups.push({ text:'+1', x:food.x*CELL+CELL/2, y:food.y*CELL, life:1 });
    elScore.textContent = score;
    elScore.classList.remove('pop'); void elScore.offsetWidth;
    elScore.classList.add('pop');
    beep(880,.06,'square',.05);
    setTimeout(()=>beep(1320,.08,'square',.05), 60);
    updateSpeedLevel();
    spawnFood();
    if (snake.length === GRID*GRID) return gameOver('You ate the entire board!', 'win');
  } else {
    snake.pop();
  }
}

function updateSpeedLevel(){
  const lv = Math.floor(score/SCORE_STEP) + 1;
  if (lv !== speedLevel){
    speedLevel = lv;
    updateHUD();
    showToast();
    screen.classList.remove('boost'); void screen.offsetWidth;
    screen.classList.add('boost');
  }
}

let toastTimer = null;
function showToast(){
  toast.classList.remove('show'); void toast.offsetWidth;
  toast.classList.add('show');
  clearTimeout(toastTimer);
  toastTimer = setTimeout(()=>toast.classList.remove('show'), 900);
}

function spawnParticles(cx, cy, color, count){
  for (let i=0; i<count; i++){
    const angle = Math.random()*Math.PI*2;
    const speed = .6 + Math.random()*2.2;
    particles.push({
      x:cx, y:cy,
      vx: Math.cos(angle)*speed,
      vy: Math.sin(angle)*speed,
      life:1,
      maxLife:.5 + Math.random()*.5,
      size:2 + Math.random()*3,
      color:color
    });
  }
}

/* ═══ Overlays & end state ═══ */
function showOverlay(variant, title, msg, scoreHTML){
  overlay.className = 'overlay show ' + variant;
  ovTitle.textContent = title;
  ovMsg.textContent  = msg;
  ovScore.innerHTML  = scoreHTML || '';
}
function hideOverlay(){ overlay.className = 'overlay'; }

function gameOver(reason, variant){
  variant = variant || 'over';
  state = 'over';
  newBest = score > best;
  if (newBest){
    best = score;
    try { localStorage.setItem('snake.best', String(best)); } catch(e){}
  }
  updateHUD();
  shake = 14;
  flash = 1;
  spawnParticles(snake[0].x*CELL + CELL/2, snake[0].y*CELL + CELL/2, '#ff6b5e', 26);
  beep(300, .55, 'sawtooth', .06, .45);
  showOverlay(variant, variant==='win' ? 'YOU WIN!' : 'GAME OVER', reason,
    '<div class="final">SCORE ' + score + '</div>' +
    (newBest ? '<div class="newbest">✦ NEW BEST!</div>' : ''));
}

function startGame(){
  reset();
  state = 'playing';
  hideOverlay();
  beep(520,.07,'square',.05);
  setTimeout(()=>beep(780,.09,'square',.05), 70);
  requestAnimationFrame(frame);
}

/* ═══ Input ═══ */
const DIRS = {
  ArrowUp:{x:0,y:-1}, ArrowDown:{x:0,y:1},
  ArrowLeft:{x:-1,y:0}, ArrowRight:{x:1,y:0},
  w:{x:0,y:-1}, s:{x:0,y:1},
  a:{x:-1,y:0}, d:{x:1,y:0}
};

window.addEventListener('keydown', e=>{
  const k = e.key;
  const isGameKey = (k === ' ' || k === 'Spacebar') || !!DIRS[k];
  if (!isGameKey) return;
  e.preventDefault();

  if (k === ' ' || k === 'Spacebar'){
    if (!e.repeat && (state === 'ready' || state === 'over')) startGame();
    return;
  }

  const d = DIRS[k];
  if (state === 'ready' || state === 'over') startGame();
  if (state !== 'over') queueDir(d);
});

/* ═══ Effects update ═══ */
function updateEffects(dt){
  for (let i=particles.length-1; i>=0; i--){
    const p = particles[i];
    p.x += p.vx*dt;
    p.y += p.vy*dt;
    p.life -= dt/p.maxLife;
    if (p.life <= 0) particles.splice(i,1);
  }
  for (let i=popups.length-1; i>=0; i--){
    const p = popups[i];
    p.life -= dt/1000;
    p.y  -= dt*.03;
    if (p.life <= 0) popups.splice(i,1);
  }
  if (shake > 0) shake = Math.max(0, shake - dt*.06);
  if (flash > 0) flash = Math.max(0, flash - dt/600);
}

/* ═══ Rendering ═══ */
function render(t){
  ctx.save();

  if (shake > 0){
    const s = shake/14;
    ctx.translate((Math.random()-.5)*s*16, (Math.random()-.5)*s*16);
  }

  drawBoard();
  drawFood(t);
  drawSnake();
  drawParticles();
  drawPopups();

  ctx.restore();

  if (flash > 0){
    ctx.fillStyle = 'rgba(255,50,60,' + (flash*.35) + ')';
    ctx.fillRect(0, 0, CANVAS_SIZE, CANVAS_SIZE);
  }
}

function drawBoard(){
  ctx.fillStyle = '#0c1f17';
  ctx.fillRect(0, 0, CANVAS_SIZE, CANVAS_SIZE);

  ctx.strokeStyle = 'rgba(94,160,120,.10)';
  ctx.lineWidth = 1;
  for (let i=0; i<=GRID; i++){
    const p = i*CELL;
    ctx.beginPath(); ctx.moveTo(p, 0);         ctx.lineTo(p, CANVAS_SIZE); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(0, p);         ctx.lineTo(CANVAS_SIZE, p); ctx.stroke();
  }

  const g = ctx.createRadialGradient(CANVAS_SIZE/2, CANVAS_SIZE/2, 80,
                                     CANVAS_SIZE/2, CANVAS_SIZE/2, CANVAS_SIZE*.78);
  g.addColorStop(0, 'rgba(0,0,0,0)');
  g.addColorStop(1, 'rgba(0,0,0,.38)');
  ctx.fillStyle = g;
  ctx.fillRect(0, 0, CANVAS_SIZE, CANVAS_SIZE);

  ctx.strokeStyle = 'rgba(126,247,165,.16)';
  ctx.strokeRect(1, 1, CANVAS_SIZE-2, CANVAS_SIZE-2);
}

function drawFood(t){
  if (food.x === -1) return;
  const cx = food.x*CELL + CELL/2;
  const cy = food.y*CELL + CELL/2;
  const pulse = Math.sin(t/280)*.5 + .5;
  const r = 4.5 + pulse*1.6;

  ctx.save();
  ctx.shadowColor = '#ff6b5e';
  ctx.shadowBlur  = 12 + pulse*7;
  ctx.fillStyle   = '#ff6b5e';
  ctx.beginPath(); ctx.arc(cx, cy, r, 0, Math.PI*2); ctx.fill();
  ctx.restore();

  ctx.fillStyle = 'rgba(255,255,255,.75)';
  ctx.beginPath(); ctx.arc(cx - r*.35, cy - r*.35, r*.32, 0, Math.PI*2); ctx.fill();
}

function drawSnake(){
  const n = snake.length;
  for (let i=n-1; i>=0; i--){
    const seg  = snake[i];
    const f    = n<=1 ? 0 : i/(n-1);
    const inset = i===0 ? 2.5 : 3;
    const size  = CELL - inset*2 - (i===n-1 ? 1 : 0);
    rr(seg.x*CELL + inset, seg.y*CELL + inset, size, size, i===0 ? 8 : 5);
    ctx.fillStyle = 'hsl(152,78%,' + (63 - f*31) + '%)';
    if (i===0){ ctx.shadowColor = 'rgba(140,240,168,.55)'; ctx.shadowBlur = 10; }
    ctx.fill();
    ctx.shadowBlur = 0;
    ctx.strokeStyle = 'rgba(5,17,12,.85)';
    ctx.lineWidth = 1.5;
    ctx.stroke();
  }
  drawEyes(dir);
}

function drawEyes(d){
  const cx = snake[0].x*CELL + CELL/2;
  const cy = snake[0].y*CELL + CELL/2;
  const px = -d.y, py = d.x;           // perpendicular to direction
  const ex1 = cx + d.x*4 + px*5,  ey1 = cy + d.y*4 + py*5;
  const ex2 = cx + d.x*4 - px*5,  ey2 = cy + d.y*4 - py*5;

  ctx.fillStyle = '#08130e';
  ctx.beginPath(); ctx.arc(ex1, ey1, 3.2, 0, Math.PI*2); ctx.fill();
  ctx.beginPath(); ctx.arc(ex2, ey2, 3.2, 0, Math.PI*2); ctx.fill();

  ctx.fillStyle = '#fff';
  ctx.beginPath(); ctx.arc(ex1 + d.x*1.2, ey1 + d.y*1.2, 1.6, 0, Math.PI*2); ctx.fill();
  ctx.beginPath(); ctx.arc(ex2 + d.x*1.2, ey2 + d.y*1.2, 1.6, 0, Math.PI*2); ctx.fill();
}

function drawParticles(){
  for (const p of particles){
    ctx.globalAlpha = p.life*p.life;
    ctx.fillStyle = p.color;
    ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
  }
  ctx.globalAlpha = 1;
}

function drawPopups(){
  ctx.font = '700 15px ui-monospace, Menlo, Consolas, monospace';
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';
  for (const p of popups){
    ctx.globalAlpha = Math.min(1, p.life*1.4);
    ctx.fillStyle = '#ffc857';
    ctx.shadowColor = '#ffc857';
    ctx.shadowBlur = 8;
    ctx.fillText(p.text, p.x, p.y);
  }
  ctx.globalAlpha = 1;
  ctx.shadowBlur = 0;
}

/* rounded-rect path helper (no reliance on ctx.roundRect) */
function rr(x, y, w, h, r){
  r = Math.min(r, w/2, h/2);
  ctx.beginPath();
  ctx.moveTo(x+r, y);
  ctx.arcTo(x+w, y,     x+w, y+h, r);
  ctx.arcTo(x+w, y+h,   x,    y+h, r);
  ctx.arcTo(x,    y+h,   x,    y,    r);
  ctx.arcTo(x,    y,     x+w, y,    r);
  ctx.closePath();
}

/* ═══ Main loop ═══ */
let lastTime = 0, acc = 0;
function frame(t){
  if (!lastTime) lastTime = t;
  const dt = Math.min(t - lastTime, 250);   // clamp after tab switches
  lastTime = t;

  if (state === 'playing'){
    acc += dt;
    let guard = 0;
    while (state === 'playing' && acc >= currentSpeed() && guard++ < 12){
      step();
      acc -= currentSpeed();
    }
  }

  updateEffects(dt);
  render(t);
  requestAnimationFrame(frame);
}

/* ═══ Boot ═══ */
reset();
updateHUD();
showOverlay('ready', 'READY', 'PRESS SPACE OR ARROW KEYS TO START');
requestAnimationFrame(frame);
