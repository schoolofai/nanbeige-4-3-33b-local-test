doTick:

```js
function doTick(){
  // consume queue
  while(queue.length){
    const d = queue[0];
    if(d.x === -dir.x && d.y === -dir.y){ queue.shift(); continue; } // reject reverse
    if(d.x===dir.x && d.y===dir.y){ queue.shift(); continue; }
    dir = d; queue.shift(); break;
  }
  prevSnake = snake.map(s=>({...s}));
  const head = {x: snake[0].x + dir.x, y: snake[0].y + dir.y};

  // walls
  if(head.x<0||head.y<0||head.x>=GRID||head.y>=GRID){ kill(); return; }

  const growing = head.x===food.x && head.y===food.y;
  const bodyToCheck = growing ? snake : snake.slice(0,-1);
  for(const s of bodyToCheck){ if(s.x===head.x&&s.y===head.y){ kill(); return; } }

  snake.unshift(head);
  if(growing){
    score += 1;
    const newLevel = Math.floor(score/5);
    if(newLevel !== level){
      level = newLevel;
      tickMs = Math.max(MIN_MS, BASE_MS * Math.pow(SPEED_DECAY, level));
      flashBanner('SPEED UP!');
      stage.classList.remove('boost'); void stage.offsetWidth; stage.classList.add('boost');
      updateHUD();
    }
    spawnFood();
    burst(food.x, food.y, '#ff5d5d');
    floaters.push({text:'+1', x:(food.x+0.5)*CELL, y:food.y*CELL, life:1, vy:-18});
    el.score.classList.remove('bump'); void el.score.offsetWidth; el.score.classList.add('bump');
    updateHUD();
  } else {
    snake.pop();
  }
}
```

```html
  <main class="cabinet">
```

```html
<script>
(() => {
  'use strict';

  /* ---------- setup ---------- */
  const GRID = 20;                 // 20x20 grid
  const CELL = 24;                 // logical px per cell (480x480 board)
  const BASE_MS = 150;             // start speed
  const SPEED_DECAY = 0.92;        // slight speedup every 5 points
  const MIN_MS = 62;

  const canvas = document.getElementById('game');
  const ctx = canvas.getContext('2d');
  const dpr = Math.min(window.devicePixelRatio || 1, 2);
  const W = GRID * CELL, H = GRID * CELL;
  canvas.width = Math.round(W * dpr);
  canvas.height = Math.round(H * dpr);
  ctx.setTransform(dpr, 0, 0, dpr, 0, 0);

  const stage   = document.getElementById('stage');
  const banner  = document.getElementById('banner');
  const overlay = document.getElementById('overlay');
  const elScore = document.getElementById('score');
  const elBest  = document.getElementById('best');
  const elSpeed = document.getElementById('speed');

  let best = 0;
  try { best = parseInt(localStorage.getItem('snake-best') || '0', 10) || 0; } catch (e) {}

  /* ---------- state ---------- */
  let state = 'ready';             // ready | playing | paused | dead | win
  let snake = [], prevSnake = [], dir = {x:1, y:0}, queue = [];
  let food = null, score = 0, level = 0, tickMs = BASE_MS;
  let acc = 0, last = performance.now();
  let newBest = false;
  let particles = [], floaters = [];

  /* ---------- helpers ---------- */
  function setState(s){ state = s; }

  function updateHUD(){
    elScore.textContent = score;
    elBest.textContent  = best;
    elSpeed.textContent = '\u00d7' + (level + 1);
  }

  function spawnFood(){
    const occupied = new Set(snake.map(s => s.x + ',' + s.y));
    const free = [];
    for (let y = 0; y < GRID; y++)
      for (let x = 0; x < GRID; x++)
        if (!occupied.has(x + ',' + y)) free.push({x, y});
    if (!free.length){ win(); return; }
    food = free[(Math.random() * free.length) | 0];
  }

  function showOverlay(kind){
    let html = '';
    if (kind === 'ready'){
      html =
        '<div class="ov-title">SNAKE</div>' +
        '<p class="ov-sub">Steer with the <b>arrow keys</b> or <b>WASD</b>.</p>' +
        '<p class="ov-sub">Eat the coral dots &mdash; the snake grows. Speed rises every <b>5 points</b>.</p>' +
        '<div class="ov-prompt"><span class="key">SPACE</span> to start</div>';
    } else if (kind === 'dead'){
      html =
        '<div class="ov-title ov-title-bad">GAME OVER</div>' +
        '<div class="final"><span class="lbl">FINAL SCORE</span><span>' + score + '</span></div>' +
        (newBest ? '<div class="newbest">\u2605 NEW BEST \u2605</div>' : '') +
        '<div class="ov-prompt"><span class="key">SPACE</span> to restart</div>';
    } else if (kind === 'win'){
      html =
        '<div class="ov-title">YOU WIN!</div>' +
        '<div class="final"><span class="lbl">FULL BOARD</span><span>' + score + '</span></div>' +
        '<div class="newbest">\u2605 PERFECT RUN \u2605</div>' +
        '<div class="ov-prompt"><span class="key">SPACE</span> to play again</div>';
    } else if (kind === 'paused'){
      html =
        '<div class="ov-title">PAUSED</div>' +
        '<p class="ov-sub">Press <b>P</b> or <b>Esc</b> to resume</p>';
    }
    overlay.innerHTML = html;
    overlay.classList.remove('hidden');
  }

  function hideOverlay(){ overlay.classList.add('hidden'); }

  function flashBanner(text){
    banner.textContent = text;
    banner.classList.remove('show');
    void banner.offsetWidth;               // restart animation
    banner.classList.add('show');
  }

  function pushDir(d){
    if (d.x === dir.x && d.y === dir.y) return;
    if (d.x === -dir.x && d.y === -dir.y) return;
    if (queue.length < 3) queue.push(d);
  }

  /* ---------- game flow ---------- */
  function resetGame(running){
    snake = [{x:9,y:10},{x:8,y:10},{x:7,y:10}];
    prevSnake = snake.map(o => ({...o}));
    dir = {x:1, y:0};
    queue = [];
    score = 0; level = 0; tickMs = BASE_MS; acc = 0;
    newBest = false;
    particles = []; floaters = [];
    spawnFood();
    updateHUD();
    setState(running ? 'playing' : 'ready');
    if (!running) showOverlay('ready');
  }

  function start(dirOverride){
    resetGame(true);
    if (dirOverride && !(dirOverride.x === -dir.x && dirOverride.y === -dir.y)) dir = dirOverride;
    hideOverlay();
  }

  function doTick(){
    // consume input queue
    while (queue.length){
      const d = queue[0];
      if (d.x === -dir.x && d.y === -dir.y){ queue.shift(); continue; }
      if (d.x === dir.x && d.y === dir.y){ queue.shift(); continue; }
      dir = d; queue.shift(); break;
    }

    prevSnake = snake.map(s => ({...s}));
    const head = {x: snake[0].x + dir.x, y: snake[0].y + dir.y};

    // wall collision -> game over
    if (head.x < 0 || head.y < 0 || head.x >= GRID || head.y >= GRID){ kill(); return; }

    const growing = head.x === food.x && head.y === food.y;
    const bodyToCheck = growing ? snake : snake.slice(0, -1);
    for (const s of bodyToCheck){
      if (s.x === head.x && s.y === head.y){ kill(); return; }
    }

    snake.unshift(head);

    if (growing){
      score += 1;
      const newLevel = Math.floor(score / 5);      // speed up every 5 points
      if (newLevel !== level){
        level = newLevel;
        tickMs = Math.max(MIN_MS, BASE_MS * Math.pow(SPEED_DECAY, level));
        flashBanner('SPEED UP!');
        stage.classList.remove('boost'); void stage.offsetWidth; stage.classList.add('boost');
        updateHUD();
      }
      spawnFood();
      burstPx((food.x + 0.5) * CELL, (food.y + 0.5) * CELL, '#ff5d5d', 14);
      floaters.push({text:'+1', x:(food.x + 0.5) * CELL, y:food.y * CELL, life:1});
      elScore.classList.remove('bump'); void elScore.offsetWidth; elScore.classList.add('bump');
      updateHUD();
    } else {
      snake.pop();
    }
  }

  function kill(){
    setState('dead');
    const isRecord = score > best;
    if (isRecord){
      best = score; newBest = true;
      try { localStorage.setItem('snake-best', String(best)); } catch (e) {}
    } else {
      newBest = false;
    }
    burstPx((snake[0].x + 0.5) * CELL, (snake[0].y + 0.5) * CELL, '#7ef29a', 26);
    stage.classList.remove('shake'); void stage.offsetWidth; stage.classList.add('shake');
    updateHUD();
    showOverlay('dead');
  }

  function win(){
    setState('win');
    updateHUD();
    showOverlay('win');
  }

  function pause(){
    if (state !== 'playing') return;
    setState('paused'); acc = 0;
    showOverlay('paused');
  }

  function resume(){
    if (state !== 'paused') return;
    setState('playing'); hideOverlay();
  }

  /* ---------- effects ---------- */
  function burstPx(px, py, color, n){
    for (let i = 0; i < n; i++){
      const a = Math.random() * Math.PI * 2;
      const sp = 2 + Math.random() * 5;
      particles.push({
        x:px, y:py,
        vx:Math.cos(a) * sp,
        vy:Math.sin(a) * sp - 1.5,
        life:1, color,
        size:1.8 + Math.random() * 2.6
      });
    }
  }

  function updateFx(dt){
    for (let i = particles.length - 1; i >= 0; i--){
      const p = particles[i];
      p.life -= dt / 650;
      if (p.life <= 0){ particles.splice(i, 1); continue; }
      p.vy += dt * 0.004;
      p.vx *= (1 - dt * 0.0008);
      p.x += p.vx * dt / 16;
      p.y += p.vy * dt / 16;
    }
    for (let i = floaters.length - 1; i >= 0; i--){
      floaters[i].life -= dt / 900;
      if (floaters[i].life <= 0) floaters.splice(i, 1);
    }
  }

  function drawFx(){
    for (const p of particles){
      ctx.globalAlpha = Math.max(0, p.life);
      ctx.fillStyle = p.color;
      ctx.fillRect(p.x - p.size / 2, p.y - p.size / 2, p.size, p.size);
    }
    ctx.globalAlpha = 1;
    ctx.textAlign = 'center';
    ctx.font = '700 15px ' + getComputedStyle(document.body).getPropertyValue('--mono');
    ctx.font = ctx.font.replace(/[\s"']/g, '') || '700 15px ui-monospace,monospace';
    for (const f of floaters){
      ctx.globalAlpha = Math.max(0, f.life);
      ctx.fillStyle = '#ffd166';
      ctx.fillText(f.text, f.x, f.y - (1 - f.life) * 30);
    }
    ctx.globalAlpha = 1;
  }

  /* ---------- rendering ---------- */
  function render(alpha, time){
    // board + checker
    ctx.fillStyle = '#0a241c';
    ctx.fillRect(0, 0, W, H);
    for (let y = 0; y < GRID; y++)
      for (let x = 0; x < GRID; x++)
        if ((x + y) % 2 === 0){ ctx.fillStyle = 'rgba(255,255,255,.028)'; ctx.fillRect(x * CELL, y * CELL, CELL, CELL); }

    // grid lines
    ctx.strokeStyle = 'rgba(126,242,154,.05)';
    ctx.lineWidth = 1;
    ctx.beginPath();
    for (let i = 1; i < GRID; i++){
      ctx.moveTo(i * CELL + .5, 0); ctx.lineTo(i * CELL + .5, H);
      ctx.moveTo(0, i * CELL + .5); ctx.lineTo(W, i * CELL + .5);
    }
    ctx.stroke();

    drawFood(time);
    drawSnake(alpha, time);
    drawFx();
  }

  function drawFood(time){
    const cx = (food.x + 0.5) * CELL, cy = (food.y + 0.5) * CELL;
    const pulse = Math.sin(time * 0.0045) * 0.5 + 0.5;
    const r = CELL * (0.28 + pulse * 0.055);

    ctx.save();
    ctx.strokeStyle = 'rgba(255,93,93,' + (0.25 + pulse * 0.2).toFixed(3) + ')';
    ctx.lineWidth = 2;
    ctx.beginPath(); ctx.arc(cx, cy, CELL * 0.46 + pulse * 3, 0, Math.PI * 2); ctx.stroke();
    ctx.restore();

    ctx.save();
    ctx.shadowColor = 'rgba(255,93,93,.95)';
    ctx.shadowBlur = 16 + pulse * 10;
    ctx.fillStyle = '#ff5d5d';
    ctx.beginPath(); ctx.arc(cx, cy, r, 0, Math.PI * 2); ctx.fill();
    ctx.shadowBlur = 0;
    ctx.fillStyle = '#fff';
    ctx.beginPath(); ctx.arc(cx - r * 0.32, cy - r * 0.32, r * 0.34, 0, Math.PI * 2); ctx.fill();
    ctx.restore();
  }

  function drawSnake(alpha, time){
    const pts = [];
    for (let i = 0; i < snake.length; i++){
      const cur = snake[i], prev = prevSnake[i] || cur;
      pts.push({
        x: (prev.x + (cur.x - prev.x) * alpha + 0.5) * CELL,
        y: (prev.y + (cur.y - prev.y) * alpha + 0.5) * CELL
      });
    }
    const n = pts.length;

    // body: tail -> head so head sits on top, hue shifting green -> deep teal
    for (let i = n - 1; i >= 0; i--){
      const t = n > 1 ? i / (n - 1) : 0;
      const hue = 150 - t * 22;
      const sat = 65 + t * 15;
      const light = 60 - t * 18;
      const r = CELL * (0.40 - t * 0.07);
      ctx.fillStyle = 'hsl(' + hue + ',' + sat + '%,' + light + '%)';
      ctx.beginPath(); ctx.arc(pts[i].x, pts[i].y, r, 0, Math.PI * 2); ctx.fill();
      ctx.strokeStyle = 'rgba(6,20,14,.35)'; ctx.lineWidth = 1.5; ctx.stroke();
      ctx.fillStyle = 'rgba(255,255,255,.10)';
      ctx.beginPath(); ctx.arc(pts[i].x - r * 0.3, pts[i].y - r * 0.3, r * 0.35, 0, Math.PI * 2); ctx.fill();
    }

    const a0 = pts[0], a1 = pts[1] || pts[0];
    const ang = Math.atan2(a1.y - a0.y, a1.x - a0.x);
    const fx = Math.cos(ang), fy = Math.sin(ang);
    const ex = -fy, ey = fx;
    const mx = a0.x + fx * CELL * 0.42, my = a0.y + fy * CELL * 0.42;

    // tongue flick (occasionally)
    if (state === 'playing' && (time % 1700) < 220){
      const len = CELL * 0.14 * (0.6 + 0.4 * Math.sin(time * 0.06));
      const tipX = mx + fx * len, tipY = my + fy * len;
      ctx.strokeStyle = '#ff5d5d'; ctx.lineWidth = 2; ctx.lineCap = 'round';
      ctx.beginPath(); ctx.moveTo(mx, my); ctx.lineTo(tipX, tipY); ctx.stroke();
      const bx = tipX + fx * len * 0.35, by = tipY + fy * len * 0.35;
      ctx.beginPath();
      ctx.moveTo(tipX, tipY); ctx.lineTo(bx + ex * len * 0.9, by + ey * len * 0.9);
      ctx.moveTo(tipX, tipY); ctx.lineTo(bx - ex * len * 0.9, by - ey * len * 0.9);
      ctx.stroke();
    }

    // eyes
    const off = CELL * 0.17, fwd = CELL * 0.14;
    for (const s of [-1, 1]){
      const cx = a0.x + fx * fwd + ex * off * s;
      const cy = a0.y + fy * fwd + ey * off * s;
      ctx.fillStyle = '#081712';
      ctx.beginPath(); ctx.arc(cx, cy, 2.9, 0, Math.PI * 2); ctx.fill();
      ctx.fillStyle = '#eafff2';
      ctx.beginPath(); ctx.arc(cx - fx * 1.1, cy - fy * 1.1, 1.6, 0, Math.PI * 2); ctx.fill();
    }
  }

  /* ---------- loop ---------- */
  function frame(now){
    const dt = Math.min(now - last, 100);
    last = now;

    if (state === 'playing'){
      acc += dt;
      let guard = 0;
      while (acc >= tickMs && state === 'playing' && guard++ < 4){
        acc -= tickMs;
        doTick();
      }
    } else {
      acc = 0;
    }

    const alpha = state === 'playing' ? Math.min(acc / tickMs, 1) : 0;
    render(alpha, now);
    updateFx(dt);

    requestAnimationFrame(frame);
  }

  /* ---------- input ---------- */
  const DIRS = {
    ArrowUp:{x:0,y:-1}, ArrowDown:{x:0,y:1},
    ArrowLeft:{x:-1,y:0}, ArrowRight:{x:1,y:0},
    KeyW:{x:0,y:-1}, KeyS:{x:0,y:1},
    KeyA:{x:-1,y:0}, KeyD:{x:1,y:0}
  };

  window.addEventListener('keydown', e => {
    const d = DIRS[e.code];
    if (d){
      e.preventDefault();
      if (state === 'ready' || state === 'dead' || state === 'win') start(d);
      else if (state === 'playing') pushDir(d);
      return;
    }
    if (e.code === 'Space'){
      e.preventDefault();
      if (state === 'ready' || state === 'dead' || state === 'win') start();
      return;
    }
    if (e.code === 'KeyP' || e.code === 'Escape'){
      if (state === 'playing') pause();
      else if (state === 'paused') resume();
    }
  });

  document.addEventListener('visibilitychange', () => {
    if (document.hidden && state === 'playing') pause();
  });

  /* ---------- ambient dust ---------- */
  for (let i = 0; i < 14; i++){
    const m = document.createElement('i');
    m.className = 'mote' + (i % 2 ? '' : ' warm');
    const sz = 3 + Math.random() * 4;
    m.style.left = Math.random() * 100 + '%';
    m.style.top = Math.random() * 100 + '%';
    m.style.width = m.style.height = sz + 'px';
    m.style.animationDuration = (9 + Math.random() * 11) + 's';
    m.style.animationDelay = (-Math.random() * 14) + 's';
    document.body.appendChild(m);
  }

  /* ---------- go ---------- */
  resetGame(false);
  requestAnimationFrame(frame);
})();
</script>
```
